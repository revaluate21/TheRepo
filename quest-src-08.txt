function renderTimeline(){

  pts.forEach((poi,i)=>{

    const el=document.createElement('article'); el.className=`timeline-item${current?' current':''}${done?' done':''}`;
    const dist=state.gps?formatDistance(distanceM(state.gps,poi)):formatClock(arrivalClock(i));

  });
}

function candidatePois(){

}
function renderNearby(){
  const origin=originForNearby(); let items=candidatePois().map(p=>({...p,_distance:distanceM(origin,p)}));
  if(state.filter!=='all') items=items.filter(p=>category(p).filter===state.filter);
  if(state.world==='day'&&state.filter!=='nightlife') items=items.filter(p=>p.category!=='nightlife');
  if(state.world==='night'&&state.filter!=='inside') items=items.filter(p=>p.id!=='estufa');
  items.sort((a,b)=>a._distance-b._distance); items=items.slice(0,12);
  const grid=$('#nearbyGrid'); grid.innerHTML='';
  items.forEach(poi=>{
    const c=category(poi), status=statusFor(poi), card=document.createElement('article'); card.className='nearby-card';
    const media=poi.photo?`<img src="${photoUrl(poi.photo,620)}" alt="${poi.name}" loading="lazy" onerror="this.hidden=true">`:`<div class="nearby-emoji">${c.icon}</div>`;
    card.innerHTML=`<div class="nearby-media" style="background:linear-gradient(135deg,${c.color},#55c8ff)">${media}<span class="nearby-distance">${formatDistance(poi._distance)}</span></div><div class="nearby-body"><div class="nearby-meta" style="color:${c.color}">${c.label} · ${status.label}</div><h3>${poi.name}</h3><p>${state.world==='day'?poi.day:poi.night}</p></div>`;
    card.addEventListener('click',()=>openDetails(poi,activeRoute().stops.indexOf(poi.id),activeRoute().stops.includes(poi.id)?'route':'side')); grid.appendChild(card);
  });
}
